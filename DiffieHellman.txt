
import java.util.*;

public class DiffieHellman {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int p = 23, g = 5;

        System.out.print("Enter private key of A: ");
        int a = sc.nextInt();

        System.out.print("Enter private key of B: ");
        int b = sc.nextInt();

        int A = 1, B = 1, keyA = 1, keyB = 1;

        for (int i = 0; i < a; i++)
            A = (A * g) % p;

        for (int i = 0; i < b; i++)
            B = (B * g) % p;

        for (int i = 0; i < a; i++)
            keyA = (keyA * B) % p;

        for (int i = 0; i < b; i++)
            keyB = (keyB * A) % p;

        System.out.println("Public key of A: " + A);
        System.out.println("Public key of B: " + B);
        System.out.println("Secret key of A: " + keyA);
        System.out.println("Secret key of B: " + keyB);

        sc.close();
    }
}
